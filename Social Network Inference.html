# Example data should have:

festival_data <- read.csv("C:/Users/delee/OneDrive/DACSS 604 Data/FestivalData.csv")





# Clean variables
festival_data <- festival_data %>%
  mutate(
    Musician = trimws(Musician),
    Instrument = trimws(Instrument),
    Festival = trimws(Festival),
    Year = as.character(Year),
    festival_year = paste(Festival, Year, sep = "_"),
    instrument_family = case_when(
      Instrument %in% c("Violin", "Viola", "Cello", "Double Bass", "Bass") ~ "Strings",
      Instrument %in% c("Flute", "Oboe", "Clarinet", "Bassoon") ~ "Woodwinds",
      Instrument %in% c("Horn", "Trumpet", "Trombone", "Tuba") ~ "Brass",
      Instrument %in% c("Percussion", "Timpani") ~ "Percussion",
      Instrument %in% c("Piano", "Keyboard", "Harp") ~ "Keyboard/Harp",
      Instrument %in% c("Conductor") ~ "Conductor",
      TRUE ~ "Other"
    )
  )

# Create incidence matrix: rows = musicians, columns = festival-years
incidence <- table(festival_data$Musician, festival_data$festival_year)
incidence[incidence > 0] <- 1

# Create two-mode network
bip_net <- network(
  incidence,
  matrix.type = "bipartite",
  directed = FALSE
)

# Confirm bipartite mode size
set.network.attribute(bip_net, "bipartite", nrow(incidence))

# Musician attributes
musician_attrs <- festival_data %>%
  group_by(Musician) %>%
  summarise(
    Instrument = first(Instrument),
    instrument_family = first(instrument_family),
    n_festival_years = n_distinct(festival_year),
    multi_festival_year = ifelse(n_festival_years > 1, 1, 0),
    .groups = "drop"
  ) %>%
  arrange(match(Musician, rownames(incidence)))

# Festival-year attributes
festival_attrs <- festival_data %>%
  group_by(festival_year) %>%
  summarise(
    Festival = first(Festival),
    Year = first(Year),
    festival_size = n_distinct(Musician),
    .groups = "drop"
  ) %>%
  arrange(match(festival_year, colnames(incidence)))

# Add musician-side attributes: mode 1
set.vertex.attribute(
  bip_net,
  "instrument_family",
  musician_attrs$instrument_family,
  v = 1:nrow(incidence)
)

set.vertex.attribute(
  bip_net,
  "n_festival_years",
  musician_attrs$n_festival_years,
  v = 1:nrow(incidence)
)

set.vertex.attribute(
  bip_net,
  "multi_festival_year",
  musician_attrs$multi_festival_year,
  v = 1:nrow(incidence)
)

# Add festival-side attributes: mode 2
festival_nodes <- (nrow(incidence) + 1):(nrow(incidence) + ncol(incidence))

set.vertex.attribute(
  bip_net,
  "Festival",
  festival_attrs$Festival,
  v = festival_nodes
)

set.vertex.attribute(
  bip_net,
  "festival_size",
  festival_attrs$festival_size,
  v = festival_nodes
)
